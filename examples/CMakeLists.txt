set(topics
    01_Primitives 02_Image 03_Viewport 04_GeometryCamera 05_MeshMaterial
    06_Light 07_ModelAnim 08_Scene 09_Text)
set(target_entries)
foreach(layer IN ITEMS Graphics RHI)
    foreach(directory IN LISTS topics)
        add_subdirectory("${layer}/${directory}")
        string(REGEX REPLACE "^[0-9]+_" "" topic "${directory}")
        if(layer STREQUAL "RHI")
            set(target "Rhi${topic}")
        else()
            set(target "Graphics${topic}")
        endif()
        set_property(TARGET ${target} PROPERTY FOLDER "Examples/${layer}")
        list(APPEND target_entries "\"${target}\":\"$<TARGET_FILE:${target}>\"")
    endforeach()
endforeach()
list(JOIN target_entries ",\n" target_json)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/ci-targets-$<CONFIG>.json" CONTENT "{\n${target_json}\n}\n")
