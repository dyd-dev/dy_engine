# Script mode synchronizes assets; normal configuration adds the examples.
if(CMAKE_SCRIPT_MODE_FILE)
    # Synchronize only the example's owned output directory.
    cmake_minimum_required(VERSION 3.20)
    if(NOT IS_DIRECTORY "${SOURCE}" OR NOT DEFINED DESTINATION OR NOT DEFINED BUILD_ROOT)
        message(FATAL_ERROR "Missing asset source, destination or build root")
    endif()
    file(REAL_PATH "${BUILD_ROOT}" build_root)
    file(REAL_PATH "${DESTINATION}" destination)
    cmake_path(IS_PREFIX build_root "${destination}" NORMALIZE inside_build)
    if(NOT inside_build OR destination STREQUAL build_root OR IS_SYMLINK "${DESTINATION}")
        message(FATAL_ERROR "Asset destination must remain inside its build directory")
    endif()
    file(MAKE_DIRECTORY "${destination}")
    file(GLOB_RECURSE inputs LIST_DIRECTORIES FALSE RELATIVE "${SOURCE}" "${SOURCE}/*")
    file(GLOB_RECURSE existing LIST_DIRECTORIES TRUE RELATIVE "${destination}" "${destination}/*")
    foreach(relative IN LISTS existing)
        if(IS_SYMLINK "${destination}/${relative}")
            message(FATAL_ERROR "Refusing to modify symlink in asset output: ${relative}")
        endif()
        file(REAL_PATH "${destination}/${relative}" resolved)
        cmake_path(IS_PREFIX destination "${resolved}" NORMALIZE inside_assets)
        if(NOT inside_assets)
            message(FATAL_ERROR "Asset output escaped its owned directory")
        endif()
        if(NOT IS_DIRECTORY "${resolved}" AND NOT relative IN_LIST inputs)
            file(REMOVE "${resolved}")
        endif()
    endforeach()
    foreach(relative IN LISTS inputs)
        if(IS_SYMLINK "${SOURCE}/${relative}")
            message(FATAL_ERROR "Symlink asset inputs are unsupported: ${relative}")
        endif()
        configure_file("${SOURCE}/${relative}" "${destination}/${relative}" COPYONLY)
    endforeach()
    return()
endif()

        if(relative MATCHES "^[.][.]/" AND NOT relative MATCHES "^[.][.]/showcase(/|$)")

function(dy_example_assets target source destination)
    if(NOT destination MATCHES "^[A-Za-z0-9_./-]+$" OR destination MATCHES "(^|/)[.][.]?(/|$)")
        message(FATAL_ERROR "Asset destination must remain relative to the example")
    endif()
    get_filename_component(input_directory "${source}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    string(MAKE_C_IDENTIFIER "${destination}" asset_name)
    if(TARGET "${target}_${asset_name}_assets")
        return()
    endif()
    if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
        set(build_root "${DY_EXAMPLES_ASSET_DIRECTORY}")
        set(output_directory "${build_root}/${target}/${destination}")
        file(GLOB_RECURSE inputs CONFIGURE_DEPENDS LIST_DIRECTORIES FALSE
            RELATIVE "${input_directory}" "${input_directory}/*")
        foreach(relative IN LISTS inputs)
            set_property(TARGET ${target} APPEND PROPERTY DY_EXAMPLE_ASSET_FILES
                "${target}/${destination}/${relative}")
            # Reuse the package's existing files in the common scene viewer.
            # Match source suffixes, since many packages contain scene.gltf.
            get_filename_component(extension "${relative}" LAST_EXT)
            string(TOLOWER "${extension}" extension)
            if(extension MATCHES "^\\.(glb|gltf|fbx|obj|hdr)$")
                file(TO_CMAKE_PATH "${input_directory}/${relative}" original)
                set_property(GLOBAL APPEND PROPERTY DY_EXAMPLE_PACKAGED_SCENES
                    "    {R\"dyf(${original})dyf\", R\"dyf(${target}/${destination}/${relative})dyf\"},\n")
            endif()
        endforeach()
    else()
        set(build_root "${CMAKE_BINARY_DIR}")
        set(output_directory "$<TARGET_FILE_DIR:${target}>/${destination}")
    endif()
    add_custom_target("${target}_${asset_name}_assets" ALL
        COMMAND "${CMAKE_COMMAND}" "-DSOURCE=${input_directory}"
            "-DDESTINATION=${output_directory}"
            "-DBUILD_ROOT=${build_root}" -P "${CMAKE_CURRENT_FUNCTION_LIST_FILE}"
        VERBATIM)
    add_dependencies(${target} "${target}_${asset_name}_assets")
endfunction()

function(dy_example_asset_file target source destination)
    if(NOT destination MATCHES "^[A-Za-z0-9_./-]+$" OR destination MATCHES "(^|/)[.][.]?(/|$)")
        message(FATAL_ERROR "Asset destination must remain relative to the example")
    endif()
    string(MAKE_C_IDENTIFIER "${destination}" asset_name)
    set(output "${DY_EXAMPLES_ASSET_DIRECTORY}/${target}/${destination}")
    get_filename_component(output_directory "${output}" DIRECTORY)
    add_custom_target("${target}_${asset_name}_assets" ALL
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${output_directory}"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${source}" "${output}"
        VERBATIM)
    add_dependencies(${target} "${target}_${asset_name}_assets")
    set_property(TARGET ${target} APPEND PROPERTY DY_EXAMPLE_ASSET_FILES "${target}/${destination}")
    get_filename_component(extension "${source}" LAST_EXT)
    string(TOLOWER "${extension}" extension)
    if(extension MATCHES "^\\.(glb|gltf|fbx|obj|hdr)$")
        file(TO_CMAKE_PATH "${source}" original)
        set_property(GLOBAL APPEND PROPERTY DY_EXAMPLE_PACKAGED_SCENES
            "    {R\"dyf(${original})dyf\", R\"dyf(${target}/${destination})dyf\"},\n")
    endif()
endfunction()

# Keep example assets current even when no executable needs relinking.
function(dy_sync_example_assets)
    set(directories "${CMAKE_CURRENT_SOURCE_DIR}")
    set(example_targets)
    set(example_modules)
    set(example_entries)
    set(example_declarations)
    set(example_asset_files)
    while(directories)
        list(POP_FRONT directories directory)
        get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
        list(APPEND directories ${children})
        get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
        foreach(target IN LISTS targets)
            get_target_property(type "${target}" TYPE)
            get_target_property(entry "${target}" DY_EXAMPLE_ENTRY)
            if((ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS") AND NOT entry)
                continue()
            elseif(NOT (ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS") AND NOT type STREQUAL "EXECUTABLE")
                continue()
            endif()
            if(target MATCHES "^Advanced(AccelerationStructures|RayQueryShadows|RayTracingReflections|RayTracingDenoising|HybridRendering)$")
                continue()
            endif()
            list(APPEND example_targets "${target}")
            get_target_property(source_directory "${target}" SOURCE_DIR)
            file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${source_directory}")
            if(relative MATCHES "^[.][.]/showcase(/|$)")
                set(category showcase)
            else()
                string(REGEX REPLACE "/.*" "" category "${relative}")
            endif()
            if(entry)
                set(module "${target}")
                set(executable "${target}")
            else()
                # Keep each standalone executable and compile the same sources
                # into an in-process entry with its own generated shader names.
                set(entry "dyfExample_${target}")
                set(module "${target}_module")
                get_target_property(sources "${target}" SOURCES)
                set(module_sources)
                foreach(source IN LISTS sources)
                    if(IS_ABSOLUTE "${source}" OR source MATCHES "^\\$<")
                        list(APPEND module_sources "${source}")
                    else()
                        list(APPEND module_sources "${source_directory}/${source}")
                    endif()
                endforeach()
                add_library("${module}" OBJECT ${module_sources})
                foreach(property IN ITEMS COMPILE_DEFINITIONS COMPILE_OPTIONS INCLUDE_DIRECTORIES
                    SYSTEM_INCLUDE_DIRECTORIES COMPILE_FEATURES CXX_STANDARD CXX_STANDARD_REQUIRED
                    CXX_EXTENSIONS LINK_DIRECTORIES LINK_OPTIONS)
                    get_target_property(value "${target}" "${property}")
                    if(NOT value STREQUAL "value-NOTFOUND")
                        set_property(TARGET "${module}" PROPERTY "${property}" "${value}")
                    endif()
                endforeach()
                get_target_property(libraries "${target}" LINK_LIBRARIES)
                if(libraries)
                    target_link_libraries("${module}" PRIVATE ${libraries})
                endif()
                target_compile_definitions("${module}" PRIVATE "main=${entry}"
                    "ShaderData=${target}ShaderData" DY_EXAMPLE_HOSTED=1)
                set_target_properties("${module}" PROPERTIES POSITION_INDEPENDENT_CODE ON FOLDER "Examples/Modules")
                # The original target owns shader generation and asset staging.
                add_dependencies("${module}" "${target}")
                set(executable "$<TARGET_FILE:${target}>")
            endif()
            list(APPEND example_modules "${module}")
            file(READ "${source_directory}/main.cpp" entry_source)
            if(NOT entry_source MATCHES "int[ \t\r\n]+main[ \t\r\n]*\\(([^)]*)\\)")
                message(FATAL_ERROR "Cannot find example entry point: ${source_directory}/main.cpp")
            endif()
            string(STRIP "${CMAKE_MATCH_1}" entry_arguments)
            if(entry_arguments STREQUAL "" OR entry_arguments STREQUAL "void")
                string(APPEND example_declarations "int ${entry}();\n")
                set(invoke "${entry}()")
            else()
                string(APPEND example_declarations "int ${entry}(int, char**);\n")
                set(invoke "${entry}(a, v)")
            endif()
            string(APPEND example_entries
                "    {\"${target}\", R\"dyf(${executable})dyf\", \"${category}\", +[](int a, char** v) { return ${invoke}; }},\n")
            foreach(assets Assets Models Textures)
                if(IS_DIRECTORY "${source_directory}/${assets}" AND NOT TARGET "${target}_${assets}_assets")
                    dy_example_assets("${target}" "${source_directory}/${assets}" "${assets}")
                endif()
            endforeach()
            if(entry)
                get_target_property(asset_files "${target}" DY_EXAMPLE_ASSET_FILES)
                if(asset_files)
                    list(APPEND example_asset_files ${asset_files})
                endif()
            endif()
        endforeach()
    endwhile()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/ExampleTargets.inc" CONTENT "${example_entries}")
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/ExampleDeclarations.inc" CONTENT "${example_declarations}")
    get_property(packaged_scenes GLOBAL PROPERTY DY_EXAMPLE_PACKAGED_SCENES)
    if(packaged_scenes)
        list(REMOVE_DUPLICATES packaged_scenes)
        list(JOIN packaged_scenes "" packaged_scene_entries)
    else()
        set(packaged_scene_entries "")
    endif()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/PackagedSceneAssets.inc" CONTENT "${packaged_scene_entries}")
    if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
        file(MAKE_DIRECTORY "${DY_EXAMPLES_ASSET_DIRECTORY}")
        list(REMOVE_DUPLICATES example_asset_files)
        list(SORT example_asset_files)
        string(REPLACE ";" "\n" asset_manifest "${example_asset_files}")
        file(WRITE "${DY_EXAMPLES_ASSET_DIRECTORY}/files.txt" "${asset_manifest}\n")
        add_library(dyf_examples STATIC "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
        set_target_properties(dyf_examples PROPERTIES POSITION_INDEPENDENT_CODE ON)
    else()
        add_executable(dyf_examples "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
        set_target_properties(dyf_examples PROPERTIES
            OUTPUT_NAME visualizer RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}>")
    endif()
    foreach(module IN LISTS example_modules)
        target_sources(dyf_examples PRIVATE "$<TARGET_OBJECTS:${module}>")
        get_target_property(libraries "${module}" LINK_LIBRARIES)
        if(libraries)
            target_link_libraries(dyf_examples PUBLIC ${libraries})
        endif()
    endforeach()
    target_sources(dyf_examples PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/Workbench.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/HostedExample.cpp")
    target_compile_features(dyf_examples PRIVATE cxx_std_17)
    target_link_libraries(dyf_examples PRIVATE dy_engine dyf::ImGui dyf::Model dyf::AdvancedRendering)
    if(DY_ENABLE_TRACY)
        target_link_libraries(dyf_examples PRIVATE dyf::Tracy)
    endif()
    if(DY_ENABLE_RENDERDOC)
        target_link_libraries(dyf_examples PRIVATE dyf::RenderDoc)
    endif()
    if(USE_D3D12)
        set(example_backend D3D12)
    elseif(USE_METAL)
        set(example_backend Metal)
    elseif(USE_VULKAN)
        set(example_backend Vulkan)
    else()
        set(example_backend Null)
    endif()
    target_compile_definitions(dyf_examples PRIVATE "DY_EXAMPLES_BACKEND=\"${example_backend}\"" DY_EXAMPLE_HOSTED=1)
    # Resolve repository assets beside the actual executable, not the launcher's
    # working directory. Mobile uses the packaged root provided by its host.
    get_filename_component(repository_root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
    file(RELATIVE_PATH repository_relative "${CMAKE_BINARY_DIR}" "${repository_root}")
    string(REPLACE "\"" "\\\"" repository_relative "${repository_relative}")
    target_compile_definitions(dyf_examples PRIVATE "DY_EXAMPLES_REPOSITORY_RELATIVE=\"${repository_relative}\"")
    if(WIN32)
        target_compile_definitions(dyf_examples PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
    target_include_directories(dyf_examples PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
    add_dependencies(dyf_examples ${example_targets})
    set_target_properties(dyf_examples PROPERTIES FOLDER Examples)
endfunction()

foreach(layer IN ITEMS dyf RHI)
    file(GLOB_RECURSE declarations CONFIGURE_DEPENDS "${layer}/*/CMakeLists.txt")
    foreach(declaration IN LISTS declarations)
        get_filename_component(directory "${declaration}" DIRECTORY)
        get_filename_component(topic "${directory}" NAME)
        if(topic STREQUAL "07_ModelAnim" AND NOT DY_EXTEND_MODEL)
            continue()
        endif()
        if(topic STREQUAL "06_Light" AND NOT DY_EXTEND_ADVANCED_RENDERING)
            continue()
        endif()
        # The separate editor creates a writable project beside its working
        # directory. Mobile hosts have not supplied a project directory yet;
        # their common ImGui workbench remains available without file writes.
        if(topic STREQUAL "10_ImGui" AND (NOT DY_EXTEND_IMGUI OR ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS"))
            continue()
        endif()
        add_subdirectory("${directory}")
        get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
        foreach(target IN LISTS targets)
            set_property(TARGET ${target} PROPERTY FOLDER "Examples/${layer}")
        endforeach()
    endforeach()
endforeach()
