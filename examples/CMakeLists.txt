if(DY_CI)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../.github/ci/ExampleSupport.cmake")
endif()
foreach(layer IN ITEMS dyf RHI)
    file(GLOB_RECURSE declarations CONFIGURE_DEPENDS "${layer}/*/CMakeLists.txt")
    foreach(declaration IN LISTS declarations)
        get_filename_component(directory "${declaration}" DIRECTORY)
        get_filename_component(topic "${directory}" NAME)
        if(topic STREQUAL "07_ModelAnim" AND NOT DY_EXTEND_MODEL)
            continue()
        endif()
        if(topic STREQUAL "10_ImGui" AND NOT DY_EXTEND_IMGUI)
            continue()
        endif()
        add_subdirectory("${directory}")
        get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
        foreach(target IN LISTS targets)
            set_property(TARGET ${target} PROPERTY FOLDER "Examples/${layer}")
        endforeach()
    endforeach()
endforeach()
add_subdirectory(advanced)
if(DY_CI)
    dy_write_example_manifest()
endif()
