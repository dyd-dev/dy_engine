cmake_minimum_required(VERSION 3.20)
project(PerformancePlayground LANGUAGES CXX)

enable_testing()
add_executable(PlaygroundControllerTests ControllerTests.cpp)
target_compile_features(PlaygroundControllerTests PRIVATE cxx_std_17)
add_test(NAME PlaygroundController COMMAND PlaygroundControllerTests)

option(PLAYGROUND_CONTROLLER_ONLY "Build portable movement checks only" OFF)
if(PLAYGROUND_CONTROLLER_ONLY)
    return()
endif()
get_filename_component(engine_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(DY_ENGINE_SOURCE_DIR "${engine_root}" CACHE PATH "dy_engine checkout")
set(DY_EXTEND_MODEL ON CACHE BOOL "" FORCE)
option(DY_ENABLE_TRACY "Enable the Tracy profiler connection" OFF)
set(DY_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
option(USE_D3D12 "Build Windows D3D12 game" ON)
add_subdirectory("${DY_ENGINE_SOURCE_DIR}" engine EXCLUDE_FROM_ALL)
add_executable(PerformancePlayground main.cpp World.cpp)
target_link_libraries(PerformancePlayground PRIVATE dyf::Model)
add_executable(PlaygroundCompatibilityTests CompatibilityTests.cpp)
target_link_libraries(PlaygroundCompatibilityTests PRIVATE dy_engine)
add_test(NAME PlaygroundCompatibility COMMAND PlaygroundCompatibilityTests)
set_tests_properties(PlaygroundCompatibility PROPERTIES TIMEOUT 60)
if(DY_ENABLE_TRACY)
    target_sources(PerformancePlayground PRIVATE TracyMemory.cpp)
    target_compile_definitions(PerformancePlayground PRIVATE PLAYGROUND_TRACY_ENABLED=1)
    target_link_libraries(PerformancePlayground PRIVATE dyf::Tracy Tracy::TracyClient)
    if(MSVC)
        target_compile_options(PerformancePlayground PRIVATE /Zi)
        target_link_options(PerformancePlayground PRIVATE /DEBUG)
    endif()
    add_executable(PlaygroundTracyMemoryTests TracyMemoryTests.cpp TracyMemory.cpp)
    target_link_libraries(PlaygroundTracyMemoryTests PRIVATE dy_engine_options Tracy::TracyClient)
    add_test(NAME PlaygroundTracyMemory COMMAND PlaygroundTracyMemoryTests)
endif()
include(FetchContent)
FetchContent_GetProperties(ufbx SOURCE_DIR playground_ufbx_dir)
target_include_directories(PerformancePlayground PRIVATE "${playground_ufbx_dir}")
add_test(NAME PlaygroundAssets COMMAND PerformancePlayground --validate-assets)
if(DY_ENABLE_TRACY)
    set_tests_properties(PlaygroundTracyMemory PlaygroundCompatibility PlaygroundAssets PROPERTIES
        ENVIRONMENT "DY_LOG_AUTO=0;TRACY_ONLY_LOCALHOST=1;TRACY_ONLY_IPV4=1")
endif()
add_custom_command(TARGET PerformancePlayground POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:PerformancePlayground>/Assets"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/ASSET_SOURCES.md" "$<TARGET_FILE_DIR:PerformancePlayground>/ASSET_SOURCES.md"
    VERBATIM)
set_target_properties(PerformancePlayground PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:PerformancePlayground>")
