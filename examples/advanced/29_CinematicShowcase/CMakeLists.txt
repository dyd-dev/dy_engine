cmake_minimum_required(VERSION 3.20)
project(AdvancedCinematicShowcase LANGUAGES CXX)

get_filename_component(engine_default "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(DY_ENGINE_SOURCE_DIR "${engine_default}" CACHE PATH "dy_engine source directory.")
if(NOT TARGET dy_engine)
    option(DY_EXTEND_MODEL "Build the optional Model extension." ON)
    option(DY_EXTEND_ADVANCED_RENDERING "Build the optional AdvancedRendering extension." ON)
    add_subdirectory("${DY_ENGINE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/dy_engine" EXCLUDE_FROM_ALL)
endif()

if(NOT TARGET dyf::Model)
    if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        message(FATAL_ERROR "CinematicShowcase requires DY_EXTEND_MODEL=ON.")
    endif()
    message(STATUS "Skipping CinematicShowcase: enable DY_EXTEND_MODEL to load environment assets.")
    return()
endif()

if(NOT TARGET dyf::AdvancedRendering)
    if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        message(FATAL_ERROR "CinematicShowcase requires DY_EXTEND_ADVANCED_RENDERING=ON.")
    endif()
    message(STATUS "Skipping CinematicShowcase: enable DY_EXTEND_ADVANCED_RENDERING for the advanced raster passes.")
    return()
endif()

set(DY_SHOWCASE_SCENE "/home/suhan/downloads/Bistro_v5_2/BistroExterior.fbx" CACHE FILEPATH "Default external glTF, GLB, FBX or OBJ environment.")
set(DY_SHOWCASE_ENVIRONMENT "/home/suhan/downloads/Bistro_v5_2/san_giuseppe_bridge_4k.hdr" CACHE FILEPATH "Default external equirectangular HDR sky (.hdr).")
set(DY_SHOWCASE_EMERALD_SCENE "/home/suhan/downloads/EmeraldSquare_v4_1/EmeraldSquare_Day.fbx" CACHE FILEPATH "Original Emerald Square daytime FBX.")
set(DY_SHOWCASE_EMERALD_ENVIRONMENT "/home/suhan/downloads/EmeraldSquare_v4_1/symmetrical_garden_4k.hdr" CACHE FILEPATH "Original Emerald Square daytime HDR environment.")
set(DY_SHOWCASE_KNIGHT_ACTOR "/home/suhan/downloads/Intel_Animated_Knight/pkg_e_knight_anim/Exports/FBX/Knight_USD_002.fbx" CACHE FILEPATH "Original Intel animated Knight FBX, selected with --actor knight.")
set(DY_SHOWCASE_COMPARISON_SCENE "/home/suhan/downloads/main_sponza/NewSponza_Main_glTF_003.gltf" CACHE FILEPATH "Original Sponza for comparison with the legacy-renderer branch.")

add_executable(AdvancedCinematicShowcase main.cpp)
target_link_libraries(AdvancedCinematicShowcase PRIVATE dyf::Model dyf::AdvancedRendering)
foreach(comparison IN ITEMS SCENE CAMERA)
    if(comparison STREQUAL "SCENE")
        set(comparison_path "${DY_SHOWCASE_COMPARISON_SCENE}")
    else()
        set(comparison_path "${CMAKE_CURRENT_SOURCE_DIR}/SponzaComparison.camera.csv")
    endif()
    get_filename_component(comparison_path "${comparison_path}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    file(TO_CMAKE_PATH "${comparison_path}" comparison_path)
    string(REPLACE "\"" "\\\"" comparison_path "${comparison_path}")
    target_compile_definitions(AdvancedCinematicShowcase PRIVATE "DY_SHOWCASE_COMPARISON_${comparison}=\"${comparison_path}\"")
endforeach()
if(EXISTS "${DY_SHOWCASE_KNIGHT_ACTOR}" AND NOT IS_DIRECTORY "${DY_SHOWCASE_KNIGHT_ACTOR}")
    get_filename_component(knight_asset "${DY_SHOWCASE_KNIGHT_ACTOR}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    file(TO_CMAKE_PATH "${knight_asset}" knight_asset)
    string(REPLACE "\"" "\\\"" knight_asset "${knight_asset}")
    target_compile_definitions(AdvancedCinematicShowcase PRIVATE "DY_SHOWCASE_KNIGHT_ACTOR=\"${knight_asset}\"")
endif()
# 실행 파일만 배포할 때도 원저자 LTC 표의 배포 조건을 함께 전달한다.
add_custom_command(TARGET AdvancedCinematicShowcase POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${DY_ENGINE_SOURCE_DIR}/src/dyf/Assets/ltc/LICENSE"
        "$<TARGET_FILE_DIR:AdvancedCinematicShowcase>/AreaLightingLtc.LICENSE" VERBATIM)
foreach(kind IN ITEMS SCENE ENVIRONMENT)
    if(NOT "${DY_SHOWCASE_EMERALD_${kind}}" STREQUAL "")
        get_filename_component(emerald "${DY_SHOWCASE_EMERALD_${kind}}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        file(TO_CMAKE_PATH "${emerald}" emerald)
        string(REPLACE "\"" "\\\"" emerald "${emerald}")
        target_compile_definitions(AdvancedCinematicShowcase PRIVATE "DY_SHOWCASE_EMERALD_${kind}=\"${emerald}\"")
    endif()
endforeach()

# 발표용 환경·대량 객체·GPU 컬링 예제가 같은 엔진 빌드를 공유한다.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../30_DenseArena" "dense-arena")
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../18_GPUCulling" "gpu-culling")
endif()

# Assets stay beside their original external buffers and textures.
if(NOT "${DY_SHOWCASE_SCENE}" STREQUAL "")
    get_filename_component(asset "${DY_SHOWCASE_SCENE}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${asset}" OR IS_DIRECTORY "${asset}")
        message(FATAL_ERROR "DY_SHOWCASE_SCENE must name an existing file: ${asset}")
    endif()
    file(TO_CMAKE_PATH "${asset}" asset)
    string(REPLACE "\"" "\\\"" asset "${asset}")
    target_compile_definitions(AdvancedCinematicShowcase PRIVATE "DY_SHOWCASE_DEFAULT_SCENE=\"${asset}\"")
endif()

if(NOT "${DY_SHOWCASE_ENVIRONMENT}" STREQUAL "")
    get_filename_component(sky "${DY_SHOWCASE_ENVIRONMENT}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${sky}" OR IS_DIRECTORY "${sky}")
        message(FATAL_ERROR "DY_SHOWCASE_ENVIRONMENT must name an existing .hdr file: ${sky}")
    endif()
    file(TO_CMAKE_PATH "${sky}" sky)
    string(REPLACE "\"" "\\\"" sky "${sky}")
    target_compile_definitions(AdvancedCinematicShowcase PRIVATE "DY_SHOWCASE_DEFAULT_ENVIRONMENT=\"${sky}\"")
endif()
