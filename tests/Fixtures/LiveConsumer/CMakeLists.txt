cmake_minimum_required(VERSION 3.20)
project(LiveExternalConsumer LANGUAGES CXX)
enable_testing()
if(NOT TARGET dy_engine)
    get_filename_component(engine_default "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    set(DY_ENGINE_SOURCE_DIR "${engine_default}" CACHE PATH "Engine source used by this independent consumer")
    set(DY_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(DY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(DY_BUILD_PROFILING_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${DY_ENGINE_SOURCE_DIR}" engine EXCLUDE_FROM_ALL)
endif()
add_executable(LiveExternalConsumer main.cpp)
target_link_libraries(LiveExternalConsumer PRIVATE dy_engine dyf::LiveClient)
add_executable(LiveExternalOptionsTests main.cpp)
target_link_libraries(LiveExternalOptionsTests PRIVATE dy_engine_options)
target_compile_definitions(LiveExternalOptionsTests PRIVATE DY_CONSUMER_OPTIONS_TEST=1)
add_test(NAME LiveExternalOptions COMMAND LiveExternalOptionsTests)
if(NOT USE_D3D12 AND NOT USE_VULKAN AND NOT USE_METAL)
    target_compile_definitions(LiveExternalConsumer PRIVATE DY_CONSUMER_NULL=1)
endif()
add_test(NAME LiveExternalRaw COMMAND LiveExternalConsumer --raw --verify)
add_test(NAME LiveExternalRenderer COMMAND LiveExternalConsumer --renderer --verify)
add_test(NAME LiveExternalCanvas COMMAND LiveExternalConsumer --canvas --verify)
add_test(NAME LiveExternalFeatures COMMAND LiveExternalConsumer --features --verify)
set_tests_properties(LiveExternalRaw LiveExternalRenderer LiveExternalCanvas LiveExternalFeatures PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77 ENVIRONMENT "DY_LOG_AUTO=0")
