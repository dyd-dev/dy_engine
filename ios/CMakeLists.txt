# Package the shared examples registry and engine for the UIKit application.
cmake_minimum_required(VERSION 3.20)
project(DyfIOSExamples LANGUAGES CXX OBJCXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "Use the iOS Xcode project or configure with CMAKE_SYSTEM_NAME=iOS.")
endif()
set(DY_ENABLE_TRACY OFF CACHE BOOL "" FORCE)
set(DY_ENABLE_RENDERDOC OFF CACHE BOOL "" FORCE)
set(DY_EXTEND_MODEL ON CACHE BOOL "" FORCE)
set(DY_EXTEND_ADVANCED_RENDERING ON CACHE BOOL "" FORCE)
set(DY_EXTEND_IMGUI ON CACHE BOOL "" FORCE)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
get_filename_component(DY_ENGINE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(DY_EXAMPLES_ASSET_DIRECTORY "${CMAKE_BINARY_DIR}/example-assets")
add_subdirectory("${DY_ENGINE_SOURCE_DIR}" engine)
add_subdirectory("${DY_ENGINE_SOURCE_DIR}/examples" examples)

# Xcode links one archive. Generate the actual static dependency closure instead
# of globbing old build products or maintaining a second dependency list.
set(pending dyf_examples)
set(visited)
set(archives)
while(pending)
    list(POP_FRONT pending target)
    if(NOT TARGET "${target}")
        continue()
    endif()
    get_target_property(alias "${target}" ALIASED_TARGET)
    if(alias)
        set(target "${alias}")
    endif()
    if(target IN_LIST visited)
        continue()
    endif()
    list(APPEND visited "${target}")
    get_target_property(type "${target}" TYPE)
    if(type STREQUAL "STATIC_LIBRARY")
        list(APPEND archives "$<TARGET_FILE:${target}>")
    endif()
    foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
        get_target_property(links "${target}" "${property}")
        if(NOT links)
            continue()
        endif()
        foreach(link IN LISTS links)
            if(link MATCHES "^\\$<(LINK_ONLY|BUILD_INTERFACE):(.+)>$")
                set(link "${CMAKE_MATCH_2}")
            endif()
            if(TARGET "${link}")
                list(APPEND pending "${link}")
            endif()
        endforeach()
    endforeach()
endwhile()
list(REMOVE_DUPLICATES archives)
list(JOIN archives "\n" archive_lines)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/ios-libraries-$<CONFIG>.txt" CONTENT "${archive_lines}\n")
