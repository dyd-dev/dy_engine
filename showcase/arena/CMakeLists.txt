cmake_minimum_required(VERSION 3.20)
project(ShowcaseArena LANGUAGES CXX)

get_filename_component(arena_engine_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT TARGET dy_engine)
    option(DY_EXTEND_MODEL "Build the optional Model extension." ON)
    option(DY_EXTEND_ADVANCED_RENDERING "Build the optional AdvancedRendering extension." ON)
    add_subdirectory("${arena_engine_root}" "${CMAKE_CURRENT_BINARY_DIR}/dy_engine" EXCLUDE_FROM_ALL)
endif()

if(NOT TARGET dyf::Model OR NOT TARGET dyf::AdvancedRendering)
    if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        message(FATAL_ERROR "ShowcaseArena requires DY_EXTEND_MODEL and DY_EXTEND_ADVANCED_RENDERING.")
    endif()
    message(STATUS "Skipping ShowcaseArena: enable DY_EXTEND_MODEL and DY_EXTEND_ADVANCED_RENDERING.")
    return()
endif()

add_executable(ShowcaseArena main.cpp)
target_link_libraries(ShowcaseArena PRIVATE dyf::Model dyf::AdvancedRendering)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_compile_definitions(ShowcaseArena PRIVATE DY_ARENA_MOBILE=1)
endif()
set_target_properties(ShowcaseArena PROPERTIES
    OUTPUT_NAME arena RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/showcase" FOLDER Showcase)
if(TARGET dyf::Tracy)
    target_link_libraries(ShowcaseArena PRIVATE dyf::Tracy)
endif()
if(TARGET dyf::RenderDoc)
    target_link_libraries(ShowcaseArena PRIVATE dyf::RenderDoc)
endif()

if(NOT DEFINED DY_ASSET_ROOT)
    get_filename_component(arena_asset_root "${arena_engine_root}/../assets" ABSOLUTE)
    set(DY_ASSET_ROOT "${arena_asset_root}" CACHE PATH "Showcase asset directory.")
endif()
set(DY_ARENA_GAME_ASSETS "${DY_ASSET_ROOT}/KayKit_Arena" CACHE PATH "KayKit asset directory for the Arena scene.")
get_filename_component(arena_assets "${DY_ARENA_GAME_ASSETS}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
include("${CMAKE_CURRENT_LIST_DIR}/../ShowcaseAssets.cmake")
dyf_prepare_arena_assets("${arena_assets}")
get_target_property(arena_mobile_entry ShowcaseArena DY_EXAMPLE_ENTRY)
if(arena_mobile_entry)
    dy_example_assets(ShowcaseArena "${arena_assets}" Assets)
    dy_example_asset_file(ShowcaseArena
        "${arena_engine_root}/src/Extends/AdvancedRendering/Assets/ltc/LICENSE"
        AreaLightingLtc.LICENSE)
    set(arena_assets Assets)
else()
    add_custom_command(TARGET ShowcaseArena POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${arena_engine_root}/src/Extends/AdvancedRendering/Assets/ltc/LICENSE"
            "$<TARGET_FILE_DIR:ShowcaseArena>/AreaLightingLtc.LICENSE" VERBATIM)
endif()
file(TO_CMAKE_PATH "${arena_assets}" arena_assets)
target_compile_definitions(ShowcaseArena PRIVATE "DY_ARENA_GAME_ASSETS=\"${arena_assets}\"")
