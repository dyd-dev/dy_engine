cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DyExecutionVisualizer LANGUAGES CXX)
endif()
if(NOT WIN32)
    message(FATAL_ERROR "dy_execution_visualizer currently supports Windows.")
endif()
option(DY_VIEWER_EMBED_TRACY "Embed the pinned Tracy profiler UI in the execution visualizer." ON)
option(DY_VIEWER_BUILD_TESTS "Build Execution Visualizer contracts" OFF)
set(DY_VIEWER_NEEDS_TRACY_SOURCE ${DY_VIEWER_EMBED_TRACY})
if(NOT TARGET dy_engine)
    get_filename_component(engine_default "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    set(DY_ENGINE_SOURCE_DIR "${engine_default}" CACHE PATH "dy_engine source directory")
    set(DY_EXTEND_IMGUI ON CACHE BOOL "" FORCE)
    set(DY_BUILD_EXECUTION_VISUALIZER OFF CACHE BOOL "" FORCE)
    add_subdirectory("${DY_ENGINE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/dy_engine" EXCLUDE_FROM_ALL)
endif()
if(NOT TARGET dy_imgui)
    message(FATAL_ERROR "dy_execution_visualizer requires DY_EXTEND_IMGUI=ON.")
endif()
if(NOT USE_D3D12 AND NOT USE_VULKAN)
    message(FATAL_ERROR "The Windows GUI needs USE_D3D12=ON or USE_VULKAN=ON; Null is for contract tests.")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/EmbeddedTracy.cmake")
add_executable(dy_execution_visualizer main.cpp Process.cpp Viewer.cpp ViewerState.cpp TracyMetrics.cpp FlowGraph.cpp FlowPlayback.cpp FlowView.cpp ControlsView.cpp ComparisonView.cpp LaunchOptions.cpp LaunchView.cpp ExternalTools.cpp AutoSession.cpp ToolBootstrap.cpp)
FetchContent_GetProperties(imgui SOURCE_DIR viewer_imgui_source)
target_sources(dy_execution_visualizer PRIVATE "${viewer_imgui_source}/misc/cpp/imgui_stdlib.cpp")
target_link_libraries(dy_execution_visualizer PRIVATE ole32 shell32 ws2_32 iphlpapi bcrypt)
target_link_libraries(dy_execution_visualizer PRIVATE dyf::LiveProfiling)
target_link_libraries(dy_execution_visualizer PRIVATE dy_imgui glfw)
target_link_libraries(dy_execution_visualizer PRIVATE dy_viewer_embedded_tracy)
target_compile_definitions(dy_execution_visualizer PRIVATE GLFW_INCLUDE_NONE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_link_options(dy_execution_visualizer PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup)
endif()
if(DY_VIEWER_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
    include("${CMAKE_CURRENT_SOURCE_DIR}/tests/EmbeddedTracyTests.cmake" OPTIONAL)
endif()
