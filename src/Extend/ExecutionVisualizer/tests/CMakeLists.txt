if(WIN32 AND MSVC AND GLFW_BUILD_WIN32 AND TARGET glfw)
    get_target_property(viewer_glfw_dir glfw SOURCE_DIR)
    get_target_property(viewer_glfw_sources glfw SOURCES)
    set(viewer_glfw_current "")
    foreach(source IN LISTS viewer_glfw_sources)
        get_filename_component(source_name "${source}" NAME)
        if(source_name STREQUAL "win32_window.c")
            get_filename_component(viewer_glfw_current "${source}" ABSOLUTE BASE_DIR "${viewer_glfw_dir}")
        endif()
    endforeach()
    if(NOT viewer_glfw_current)
        message(FATAL_ERROR "GLFW ownership regression requires the actual target win32_window.c")
    endif()
    foreach(version IN ITEMS Current Original)
        if(version STREQUAL "Current")
            set(modifier_source "${viewer_glfw_current}")
        else()
            set(modifier_source "${viewer_glfw_dir}/win32_window.c")
        endif()
        file(READ "${modifier_source}" modifier_text)
        string(FIND "${modifier_text}" "    handle = GetActiveWindow();" modifier_begin)
        string(FIND "${modifier_text}" "    window = _glfw.win32.disabledCursorWindow;" modifier_end)
        if(modifier_begin LESS 0 OR modifier_end LESS modifier_begin)
            message(FATAL_ERROR "GLFW modifier block changed; review ownership regression extraction")
        endif()
        math(EXPR modifier_size "${modifier_end}-${modifier_begin}")
        string(SUBSTRING "${modifier_text}" ${modifier_begin} ${modifier_size} modifier_block)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Glfw${version}Modifiers.inc" "${modifier_block}")
    endforeach()
    add_executable(ViewerGlfwWindowOwnershipTests GlfwWindowOwnershipTests.cpp)
    target_compile_features(ViewerGlfwWindowOwnershipTests PRIVATE cxx_std_17)
    target_include_directories(ViewerGlfwWindowOwnershipTests PRIVATE "${viewer_glfw_dir}/../include" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(ViewerGlfwWindowOwnershipTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN GLFW_INCLUDE_NONE)
    target_compile_options(ViewerGlfwWindowOwnershipTests PRIVATE /utf-8 /W4)
    add_test(NAME ViewerGlfwWindowOwnershipContracts COMMAND ViewerGlfwWindowOwnershipTests)
    set_tests_properties(ViewerGlfwWindowOwnershipContracts PROPERTIES TIMEOUT 20)
endif()
